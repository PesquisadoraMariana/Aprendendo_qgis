# ====================== ABRIR AS BANDAS ======================
def ler_banda(caminho):
    ds = gdal.Open(caminho)
    banda = ds.GetRasterBand(1).ReadAsArray().astype(np.float32)
    return banda

nir_2019 = ler_banda(b08_2019)

red_2026 = ler_banda(b04_2026)
nir_2026 = ler_banda(b08_2026)
